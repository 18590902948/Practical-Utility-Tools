

从 OSZICAR 中提取含 T= 行的第 1、3、5 列并保存到 energy.txt


Linux:

# 1. 过滤含"T="的行，提取第1、3、5列，输出到energy.txt,一般用这个！！！！！！！！！！！！！

grep "T=" OSZICAR | awk '{print $1, $3, $5}' > energy.txt

grep "T=" OSZICAR | awk '{print $1, $3, $7}' > energy2.txt

grep "T=" OSZICAR | awk '{print $1, $3, $9}' > energy3.txt                    

grep "T=" OSZICAR | awk '{print $1, $3, $5, $9}' OFS='\t' > energy3.txt              # 一般取这个！！！E0







Windows:

1.CMD


# 1. 过滤含"T="的行，提取第1、3、5列并写入energy.txt

for /f "tokens=1,3,5" %a in ('findstr "T=" OSZICAR') do @echo %a %b %c >> energy.txt


# 提取含"T="行的第5列，写入energy.txt
for /f "tokens=5" %a in ('findstr "T=" OSZICAR') do @echo %a >> energy.txt


2.Powershell


# 1. 匹配含"T="的行，拆分列后提取第1/3/5列，保存到energy.txt（ASCII编码兼容Linux）

Select-String -Path OSZICAR -Pattern "T=" | ForEach-Object {
    $parts = $_.Line.Split()
    Write-Output "$($parts[0]) $($parts[2]) $($parts[4])"
} | Out-File -FilePath energy.txt -Encoding ASCII



# 提取含"T="行的第5列，保存到energy.txt（ASCII编码）
Select-String -Path OSZICAR -Pattern "T=" | ForEach-Object {
    $parts = $_.Line.Split()
    Write-Output $parts[4]
} | Out-File -FilePath energy.txt -Encoding ASCII
